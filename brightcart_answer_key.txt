Key SQL:
  WITH rev AS (SELECT o.order_id, o.customer_id, SUM(i.qty*i.unit_price) v
               FROM o_clean o JOIN order_items i ON i.order_id=o.order_id
               WHERE o.status IN ('delivered','shipped') GROUP BY o.order_id)
  SELECT r.customer_id, COUNT(*) n, SUM(v) at_risk
  FROM rev r LEFT JOIN payments p ON p.order_id=r.order_id
  WHERE p.order_id IS NULL
     OR julianday(p.pay_date)-julianday((SELECT order_date FROM o_clean
          WHERE order_id=r.order_id)) > 30
  GROUP BY r.customer_id ORDER BY at_risk DESC LIMIT 5;
